import ffmpeg

def compress_video(input_path, output_path, crf=23, preset='slow'):
    """
    Compresses a video to WebM format while maintaining high quality.
    
    Parameters:
    - input_path: Path to your original video
    - output_path: Path for the output compressed video
    - crf: Quality setting (lower = better quality, 18-28 range for WebM)
    - preset: Compression speed (slower = better compression)
    """
    try:
        (
            ffmpeg
            .input(input_path)
            .output(
                output_path,
                vcodec='libvpx-vp9',  # VP9 codec for WebM
                crf=crf,
                bitrate='0',
                preset=preset,
                acodec='libopus'  # Audio codec
            )
            .run()
        )
        print(f"Successfully compressed to {output_path}")
    except ffmpeg.Error as e:
        print('Error:', e)

# Example usage:
compress_video('input.mp4', 'output.webm', crf=23, preset='slow')
🔹 Notes:

crf=23 gives almost visually lossless quality.
(Lower it to 20 or 18 for even better but bigger files.)

preset='slow' is better quality compression; use 'veryslow' if you want the smallest size.

It keeps audio too!

Output file will be WebM, optimized for the web.

⚡ Summary recommendation:

Full videos: use WebM (VP9) or MP4 (H.264)

Short animations: Animated WebP

import ffmpeg

def compress_to_mp4(input_path, output_path, crf=23, preset='slow'):
    """
    Compress video into MP4 format with very high quality.
    
    Parameters:
    - input_path: Source video file path
    - output_path: Destination MP4 file path
    - crf: Constant Rate Factor (lower = better quality, 18-28 is typical)
